/* 00_Setup.sql - databases, options, schemas, partition function */
USE master;
GO
IF DB_ID(N'HRMS_Master') IS NULL
 CREATE DATABASE HRMS_Master COLLATE Latin1_General_CI_AS
 ON PRIMARY (NAME=N'HRMS_Master_sys', FILENAME=N'$(DataPath)\HRMS_Master_sys.mdf', SIZE=64MB, FILEGROWTH=64MB),
 FILEGROUP FG_DATA DEFAULT (NAME=N'HRMS_Master_data', FILENAME=N'$(DataPath)\HRMS_Master_data.ndf', SIZE=256MB, FILEGROWTH=128MB)
 LOG ON (NAME=N'HRMS_Master_log', FILENAME=N'$(LogPath)\HRMS_Master_log.ldf', SIZE=128MB, FILEGROWTH=64MB);
GO
IF DB_ID(N'HRMS_Live') IS NULL
 CREATE DATABASE HRMS_Live COLLATE Latin1_General_CI_AS
 ON PRIMARY (NAME=N'HRMS_Live_sys', FILENAME=N'$(DataPath)\HRMS_Live_sys.mdf', SIZE=64MB, FILEGROWTH=64MB),
 FILEGROUP FG_DATA DEFAULT (NAME=N'HRMS_Live_data', FILENAME=N'$(DataPath)\HRMS_Live_data.ndf', SIZE=2GB, FILEGROWTH=512MB)
 LOG ON (NAME=N'HRMS_Live_log', FILENAME=N'$(LogPath)\HRMS_Live_log.ldf', SIZE=512MB, FILEGROWTH=256MB);
GO
DECLARE @db SYSNAME, @sql NVARCHAR(MAX);
DECLARE c CURSOR LOCAL FOR SELECT name FROM sys.databases WHERE name IN (N'HRMS_Master',N'HRMS_Live');
OPEN c; FETCH c INTO @db;
WHILE @@FETCH_STATUS=0
BEGIN
 SET @sql=N'ALTER DATABASE '+QUOTENAME(@db)+N' SET RECOVERY FULL, PAGE_VERIFY CHECKSUM, AUTO_CREATE_STATISTICS ON, AUTO_UPDATE_STATISTICS ON, AUTO_UPDATE_STATISTICS_ASYNC ON, ALLOW_SNAPSHOT_ISOLATION ON, READ_COMMITTED_SNAPSHOT ON, TRUSTWORTHY OFF, DB_CHAINING OFF WITH ROLLBACK IMMEDIATE';
 EXEC(@sql);
 SET @sql=N'ALTER DATABASE '+QUOTENAME(@db)+N' SET QUERY_STORE = ON';
 EXEC(@sql);
 FETCH c INTO @db;
END
CLOSE c; DEALLOCATE c;
GO

/* Monthly partitions for the financial year (attendance + punches). Rebuilt at year-end. */
IF NOT EXISTS (SELECT 1 FROM sys.partition_functions WHERE name='pf_Month')
BEGIN
 DECLARE @fy DATE='$(FYStart)', @i INT=1, @b NVARCHAR(MAX)=N'';
 WHILE @i<=11
 BEGIN
  SET @b+=CASE WHEN @i>1 THEN N',' ELSE N'' END+QUOTENAME(CONVERT(CHAR(10),DATEADD(MONTH,@i,@fy),23),'''');
  SET @i+=1;
 END
 EXEC(N'CREATE PARTITION FUNCTION pf_Month(DATE) AS RANGE RIGHT FOR VALUES ('+@b+N')');
 EXEC(N'CREATE PARTITION SCHEME ps_Month AS PARTITION pf_Month ALL TO ([FG_DATA])');
END
GO
