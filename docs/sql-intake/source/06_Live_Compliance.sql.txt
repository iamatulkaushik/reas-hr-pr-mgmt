/* 06_Live_Compliance.sql - HRMS_Live, schema compliance */
USE HRMS_Live;
GO
CREATE TABLE compliance.EPFContribution(
 Id BIGINT IDENTITY PRIMARY KEY, RunId INT NOT NULL REFERENCES payroll.PayrollRun, EmpId INT NOT NULL, FactoryId INT NOT NULL,
 PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, GrossWage DECIMAL(14,2) NOT NULL, EPFWage DECIMAL(14,2) NOT NULL, EPSWage DECIMAL(14,2) NOT NULL, EDLIWage DECIMAL(14,2) NOT NULL,
 EmployeeShare DECIMAL(12,2) NOT NULL, EPSShare DECIMAL(12,2) NOT NULL, EmployerEPFShare DECIMAL(12,2) NOT NULL, EDLI DECIMAL(12,2) NOT NULL, AdminCharge DECIMAL(12,2) NOT NULL,
 NCPDays DECIMAL(4,1) NOT NULL DEFAULT 0, RefundOfAdvance DECIMAL(12,2) NOT NULL DEFAULT 0, CONSTRAINT UQ_EPFC UNIQUE(RunId,EmpId));
CREATE TABLE compliance.EPFReturn(
 ReturnId INT IDENTITY PRIMARY KEY, FactoryId INT NOT NULL, PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, MemberCount INT, TotalWages DECIMAL(16,2),
 TRRN VARCHAR(30), FilePath NVARCHAR(400), Status VARCHAR(10) NOT NULL DEFAULT 'Prepared', FiledOn DATE, FiledBy SYSNAME, CONSTRAINT UQ_EPFR UNIQUE(FactoryId,PayYear,PayMonth));

CREATE TABLE compliance.ESIContribution(
 Id BIGINT IDENTITY PRIMARY KEY, RunId INT NOT NULL REFERENCES payroll.PayrollRun, EmpId INT NOT NULL, FactoryId INT NOT NULL,
 PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, ESIWage DECIMAL(14,2) NOT NULL, EmployeeShare DECIMAL(12,2) NOT NULL, EmployerShare DECIMAL(12,2) NOT NULL,
 DaysWorked DECIMAL(4,1) NOT NULL, PeriodStart DATE NOT NULL, ReasonCode TINYINT NOT NULL DEFAULT 0, CONSTRAINT UQ_ESIC UNIQUE(RunId,EmpId));
CREATE TABLE compliance.ESIReturn(
 ReturnId INT IDENTITY PRIMARY KEY, FactoryId INT NOT NULL, PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, IPCount INT, TotalWages DECIMAL(16,2),
 ChallanNo VARCHAR(30), FilePath NVARCHAR(400), Status VARCHAR(10) NOT NULL DEFAULT 'Prepared', FiledOn DATE, FiledBy SYSNAME, CONSTRAINT UQ_ESIR UNIQUE(FactoryId,PayYear,PayMonth));

CREATE TABLE compliance.PTDeduction(
 Id BIGINT IDENTITY PRIMARY KEY, RunId INT NOT NULL REFERENCES payroll.PayrollRun, EmpId INT NOT NULL, FactoryId INT NOT NULL, StateId INT NOT NULL,
 PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, GrossWage DECIMAL(14,2) NOT NULL, Amount DECIMAL(10,2) NOT NULL, CONSTRAINT UQ_PTD UNIQUE(RunId,EmpId));
CREATE TABLE compliance.PTReturn(
 ReturnId INT IDENTITY PRIMARY KEY, FactoryId INT NOT NULL, StateId INT NOT NULL, PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, TotalTax DECIMAL(14,2), ChallanNo VARCHAR(30),
 Status VARCHAR(10) NOT NULL DEFAULT 'Prepared', FiledOn DATE);

CREATE TABLE compliance.LWFDeduction(
 Id BIGINT IDENTITY PRIMARY KEY, RunId INT NOT NULL REFERENCES payroll.PayrollRun, EmpId INT NOT NULL, FactoryId INT NOT NULL, StateId INT NOT NULL,
 PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, EmployeeAmt DECIMAL(8,2) NOT NULL, EmployerAmt DECIMAL(8,2) NOT NULL, CONSTRAINT UQ_LWFD UNIQUE(RunId,EmpId));
CREATE TABLE compliance.LWFReturn(
 ReturnId INT IDENTITY PRIMARY KEY, FactoryId INT NOT NULL, StateId INT NOT NULL, PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, TotalAmount DECIMAL(14,2),
 ChallanNo VARCHAR(30), Status VARCHAR(10) NOT NULL DEFAULT 'Prepared', FiledOn DATE);

CREATE TABLE compliance.TDSProjection(
 EmpId INT NOT NULL, FYCode VARCHAR(10) NOT NULL, Regime CHAR(3) NOT NULL, ProjectedIncome DECIMAL(14,2) NOT NULL, Deductions DECIMAL(14,2) NOT NULL DEFAULT 0,
 TaxableIncome DECIMAL(14,2) NOT NULL, AnnualTax DECIMAL(14,2) NOT NULL, MonthlyTDS DECIMAL(12,2) NOT NULL, ComputedBy SYSNAME NOT NULL DEFAULT ORIGINAL_LOGIN(),
 ComputedOn DATETIME2(0) NOT NULL DEFAULT SYSUTCDATETIME(), PRIMARY KEY(EmpId,FYCode));
CREATE TABLE compliance.TDSDeduction(
 Id BIGINT IDENTITY PRIMARY KEY, RunId INT NOT NULL REFERENCES payroll.PayrollRun, EmpId INT NOT NULL, FactoryId INT NOT NULL, PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL,
 Amount DECIMAL(12,2) NOT NULL, CONSTRAINT UQ_TDSD UNIQUE(RunId,EmpId));
CREATE TABLE compliance.Form16(
 Id INT IDENTITY PRIMARY KEY, EmpId INT NOT NULL, FYCode VARCHAR(10) NOT NULL, GrossSalary DECIMAL(14,2), TaxDeducted DECIMAL(14,2), CertificateNo VARCHAR(30),
 FilePath NVARCHAR(400), IssuedOn DATE, IssuedBy SYSNAME, CONSTRAINT UQ_F16 UNIQUE(EmpId,FYCode));

CREATE TABLE compliance.Challan(
 ChallanId INT IDENTITY PRIMARY KEY, FactoryId INT NOT NULL, ChallanType VARCHAR(4) NOT NULL CHECK (ChallanType IN ('EPF','ESI','PT','LWF','TDS')),
 PayMonth TINYINT NOT NULL, PayYear SMALLINT NOT NULL, Amount DECIMAL(16,2) NOT NULL, DueDate DATE NULL, PaidDate DATE NULL, ReferenceNo VARCHAR(40),
 Status VARCHAR(10) NOT NULL DEFAULT 'Due' CHECK (Status IN ('Due','Paid','Late')), InterestPenalty DECIMAL(12,2) NOT NULL DEFAULT 0,
 CONSTRAINT UQ_Challan UNIQUE(FactoryId,ChallanType,PayYear,PayMonth));
CREATE TABLE compliance.ChallanLine(
 ChallanLineId INT IDENTITY PRIMARY KEY, ChallanId INT NOT NULL REFERENCES compliance.Challan, AccountHead VARCHAR(20) NOT NULL, Amount DECIMAL(16,2) NOT NULL);

CREATE TABLE compliance.FilingCalendar(
 CalendarId INT IDENTITY PRIMARY KEY, Code VARCHAR(10) NOT NULL UNIQUE, Name NVARCHAR(120) NOT NULL, Frequency VARCHAR(10) NOT NULL, DueDay TINYINT NOT NULL, DueMonthOffset TINYINT NOT NULL DEFAULT 1,
 Authority NVARCHAR(80), Notes NVARCHAR(200));
CREATE TABLE compliance.FilingStatus(
 FilingId INT IDENTITY PRIMARY KEY, CalendarId INT NOT NULL REFERENCES compliance.FilingCalendar, FactoryId INT NOT NULL, PeriodMonth TINYINT NOT NULL, PeriodYear SMALLINT NOT NULL,
 DueDate DATE NOT NULL, FiledOn DATE NULL, AckNo VARCHAR(40), FiledBy SYSNAME, Status VARCHAR(10) NOT NULL DEFAULT 'Pending');

CREATE TABLE compliance.MinWageCheck(
 CheckId BIGINT IDENTITY PRIMARY KEY, RunId INT NOT NULL REFERENCES payroll.PayrollRun, EmpId INT NOT NULL, PaidDays DECIMAL(4,1), WagePerDay DECIMAL(12,2), MinDailyRate DECIMAL(12,2),
 Shortfall DECIMAL(12,2), CheckedOn DATETIME2(0) NOT NULL DEFAULT SYSUTCDATETIME());

CREATE TABLE compliance.FactoryWorkerRegister(        -- Form 12 style adult worker register
 Id BIGINT IDENTITY PRIMARY KEY, EmpId INT NOT NULL, WeekStart DATE NOT NULL, WeeklyHours DECIMAL(5,1) NOT NULL, WeeklyHolidayDate DATE NULL, ExceedsLimit BIT NOT NULL DEFAULT 0, CONSTRAINT UQ_FWR UNIQUE(EmpId,WeekStart));
CREATE TABLE compliance.OvertimeRegister(
 Id BIGINT IDENTITY PRIMARY KEY, EmpId INT NOT NULL, QuarterStart DATE NOT NULL, OTHours DECIMAL(6,1) NOT NULL, ExceedsLimit BIT NOT NULL DEFAULT 0, CONSTRAINT UQ_OTR UNIQUE(EmpId,QuarterStart));
CREATE TABLE compliance.AccidentRegister(
 AccidentId INT IDENTITY PRIMARY KEY, FactoryId INT NOT NULL, EmpId INT NULL, AccidentDate DATETIME2(0) NOT NULL, Location NVARCHAR(100), Description NVARCHAR(1000),
 Severity VARCHAR(10) NOT NULL CHECK (Severity IN ('NearMiss','Minor','Major','Fatal')), LostDays INT NOT NULL DEFAULT 0, ReportedToAuthority BIT NOT NULL DEFAULT 0, ReportedOn DATE NULL);
CREATE TABLE compliance.AdolescentCertificate(
 CertId INT IDENTITY PRIMARY KEY, EmpId INT NOT NULL, CertNo VARCHAR(40) NOT NULL, IssuedOn DATE NOT NULL, ValidTo DATE NOT NULL, IssuedBy NVARCHAR(100));
CREATE TABLE compliance.ContractorLicence(
 LicenceId INT IDENTITY PRIMARY KEY, ContractorId INT NOT NULL, LicenceNo VARCHAR(40) NOT NULL, ValidFrom DATE NOT NULL, ValidTo DATE NOT NULL, MaxWorkers INT);
CREATE TABLE compliance.EstablishmentLicence(
 LicenceId INT IDENTITY PRIMARY KEY, FactoryId INT NOT NULL, LicenceType VARCHAR(20) NOT NULL, LicenceNo VARCHAR(50), ValidTo DATE, RenewalRemindDays INT NOT NULL DEFAULT 60);
CREATE TABLE compliance.InspectionNotice(
 InspectionId INT IDENTITY PRIMARY KEY, FactoryId INT NOT NULL, Authority NVARCHAR(100), InspectionDate DATE NOT NULL, Findings NVARCHAR(1000), ActionDue DATE NULL, Status VARCHAR(10) NOT NULL DEFAULT 'Open');
CREATE TABLE compliance.RuleChangeLog(                 -- legal updates applied to rate tables
 ChangeId INT IDENTITY PRIMARY KEY, Area VARCHAR(30) NOT NULL, Description NVARCHAR(500) NOT NULL, EffectiveFrom DATE NULL, SourceRef NVARCHAR(300),
 EnteredBy SYSNAME NOT NULL DEFAULT ORIGINAL_LOGIN(), EnteredOn DATETIME2(0) NOT NULL DEFAULT SYSUTCDATETIME());
GO
