/* ---- row-level security: operators see only their factories (sec.UserFactory) ---- */
CREATE FUNCTION sec.fn_FactoryFilter(@FactoryId INT) RETURNS TABLE WITH SCHEMABINDING AS
RETURN SELECT 1 AS ok WHERE IS_MEMBER('role_associate')=1 OR IS_MEMBER('role_ca')=1 OR IS_MEMBER('db_owner')=1 OR IS_SRVROLEMEMBER('sysadmin')=1
 OR EXISTS (SELECT 1 FROM sec.UserFactory u WHERE u.LoginName=USER_NAME() AND (u.FactoryId=0 OR u.FactoryId=@FactoryId));
GO
CREATE SECURITY POLICY sec.pol_Employee
 ADD FILTER PREDICATE sec.fn_FactoryFilter(FactoryId) ON hr.Employee,
 ADD BLOCK PREDICATE sec.fn_FactoryFilter(FactoryId) ON hr.Employee AFTER INSERT,
 ADD BLOCK PREDICATE sec.fn_FactoryFilter(FactoryId) ON hr.Employee AFTER UPDATE WITH (STATE=ON);
GO
/* ---- maker-checker guard: only role_associate may set approval fields ---- */
CREATE PROCEDURE sec.usp_CreateApprovalGuard @Schema SYSNAME,@Table SYSNAME,@HasStatus BIT=1 AS
BEGIN
 DECLARE @t NVARCHAR(300)=QUOTENAME(@Schema)+N'.'+QUOTENAME(@Table),@trg NVARCHAR(300)=QUOTENAME(@Schema)+N'.'+QUOTENAME(N'trg_'+@Table+N'_approval');
 DECLARE @sql NVARCHAR(MAX)=N'CREATE OR ALTER TRIGGER '+@trg+N' ON '+@t+N' AFTER INSERT,UPDATE AS
BEGIN
 SET NOCOUNT ON;
 IF IS_MEMBER(''role_associate'')=1 OR IS_MEMBER(''db_owner'')=1 OR IS_SRVROLEMEMBER(''sysadmin'')=1 RETURN;
 IF EXISTS(SELECT 1 FROM inserted WHERE ApprovedBy IS NOT NULL'+CASE WHEN @HasStatus=1 THEN N' OR Status IN (''Approved'',''Applied'')' ELSE N'' END+N')
  THROW 50500,''Only the Associate can approve. Maker-checker rule.'',1;
END';
 EXEC sys.sp_executesql @sql;
END
GO
EXEC sec.usp_CreateApprovalGuard 'hr','Promotion';
EXEC sec.usp_CreateApprovalGuard 'hr','Increment';
GO
/* ---- user directory + provisioning (run as sysadmin) ---- */
CREATE PROCEDURE sec.usp_AddUser @LoginName SYSNAME,@Role SYSNAME,@FullName NVARCHAR(150),@Password NVARCHAR(128)=NULL,@FactoryIds VARCHAR(200)='0' AS
BEGIN
 SET NOCOUNT ON;
 IF @Role NOT IN ('role_developer','role_associate','role_operator','role_ca') THROW 50510,'Unknown role.',1;
 IF IS_SRVROLEMEMBER('sysadmin')<>1 THROW 50511,'Run as sysadmin.',1;
 DECLARE @sql NVARCHAR(MAX);
 IF SUSER_ID(@LoginName) IS NULL
 BEGIN
  SET @sql=CASE WHEN @Password IS NULL THEN N'CREATE LOGIN '+QUOTENAME(@LoginName)+N' FROM WINDOWS WITH DEFAULT_DATABASE=HRMS_Live'
       ELSE N'CREATE LOGIN '+QUOTENAME(@LoginName)+N' WITH PASSWORD=N'''+REPLACE(@Password,'''','''''')+N''', CHECK_POLICY=ON, CHECK_EXPIRATION=ON, DEFAULT_DATABASE=HRMS_Live' END;
  EXEC(@sql);
 END
 SET @sql=N'IF USER_ID(N'''+REPLACE(@LoginName,'''','''''')+N''') IS NULL CREATE USER '+QUOTENAME(@LoginName)+N' FOR LOGIN '+QUOTENAME(@LoginName)+N'; ALTER ROLE '+QUOTENAME(@Role)+N' ADD MEMBER '+QUOTENAME(@LoginName)+N';';
 EXEC(N'USE HRMS_Master; '+@sql); EXEC(N'USE HRMS_Live; '+@sql);
 IF NOT EXISTS (SELECT 1 FROM sec.AppUser WHERE LoginName=@LoginName) INSERT sec.AppUser(LoginName,RoleName,FullName) VALUES(@LoginName,@Role,@FullName);
 ELSE UPDATE sec.AppUser SET RoleName=@Role,FullName=@FullName,IsActive=1 WHERE LoginName=@LoginName;
 DELETE FROM sec.UserFactory WHERE LoginName=@LoginName;
 INSERT sec.UserFactory(LoginName,FactoryId) SELECT @LoginName,CAST(value AS INT) FROM STRING_SPLIT(@FactoryIds,',') WHERE LTRIM(RTRIM(value))<>'';
 EXEC HRMS_Live.sec.usp_SyncUserFactory;
END
GO
CREATE PROCEDURE sec.usp_DisableUser @LoginName SYSNAME AS
BEGIN
 SET NOCOUNT ON;
 IF IS_SRVROLEMEMBER('sysadmin')<>1 THROW 50512,'Run as sysadmin.',1;
 EXEC(N'ALTER LOGIN '+N'['+@LoginName+N'] DISABLE');
 UPDATE sec.AppUser SET IsActive=0 WHERE LoginName=@LoginName;
 DELETE FROM sec.UserFactory WHERE LoginName=@LoginName; EXEC HRMS_Live.sec.usp_SyncUserFactory;
END
GO

/* ---- server audit (logins, role changes). Wrapped so a failure never aborts deployment. ---- */
USE master;
GO
BEGIN TRY
 EXEC(N'CREATE SERVER AUDIT HRMS_Audit TO FILE (FILEPATH=N''$(AuditPath)\'', MAXSIZE=100 MB, MAX_ROLLOVER_FILES=50, RESERVE_DISK_SPACE=OFF) WITH (ON_FAILURE=CONTINUE)');
 EXEC(N'ALTER SERVER AUDIT HRMS_Audit WITH (STATE=ON)');
 EXEC(N'CREATE SERVER AUDIT SPECIFICATION HRMS_Server_Spec FOR SERVER AUDIT HRMS_Audit ADD (FAILED_LOGIN_GROUP),ADD (SUCCESSFUL_LOGIN_GROUP),ADD (SERVER_ROLE_MEMBER_CHANGE_GROUP),ADD (SERVER_PRINCIPAL_CHANGE_GROUP),ADD (LOGIN_CHANGE_PASSWORD_GROUP) WITH (STATE=ON)');
 EXEC(N'USE HRMS_Live; CREATE DATABASE AUDIT SPECIFICATION HRMS_Live_Spec FOR SERVER AUDIT HRMS_Audit ADD (SCHEMA_OBJECT_CHANGE_GROUP),ADD (DATABASE_ROLE_MEMBER_CHANGE_GROUP),ADD (DATABASE_PERMISSION_CHANGE_GROUP),ADD (SELECT ON payroll.EmployeeSalary BY public) WITH (STATE=ON)');
END TRY
BEGIN CATCH
